async function procesujNagrodeKompilacyjna(adresKlienta, linkJSONzPinaty) {
  try {
    console.log(`[1/4] Rozpoczynam mintowanie NFT dla: ${adresKlienta}`);
    
    // RZECZYWISTY MINT NFT
    const nftTx = await mintTo({
      contract: nftContract,
      account: poolAccount,
      to: adresKlienta,
      metadata: linkJSONzPinaty, // np. "ipfs://Qm..."
    });

    console.log(`[2/4] NFT Zmintowane. Przelewam token bonusowy...`);
    
    // TRANSFER TOKENU BONUSOWEGO Z POOLA
    const tokenTx = await transferERC20({
      contract: bonusTokenContract,
      account: poolAccount,
      to: adresKlienta,
      amount: "100", // Ilość Twojego tokena bonusowego
    });

    console.log(`[3/4] Wysyłam bonusowe ETH z poola...`);
    
    // OPCJONALNIE: TRANSFER NATIVE ETH Z POOLA (np. 0.001 ETH jako bonus na start)
    const ethTx = await sendTransaction({
      account: poolAccount,
      transaction: {
        to: adresKlienta,
        value: parseEther("0.001"), // Wartość w ETH
        chain: chainId,
      },
    });

    console.log(`[4/4] Wysyłam bonusowe USDC (USD) z poola...`);
    
    // OPCJONALNIE: TRANSFER USDC (DOLARA) Z POOLA (np. 1 USDC)
    // USDC ma 6 miejsc po przecinku na Base, więc "1000000" to równowartość 1.00 USD
    const usdcTx = await prepareContractCall({
      contract: usdcContract,
      method: "function transfer(address to, uint256 value) public returns (bool)",
      params: [adresKlienta, 1000000n], 
    });
    await sendTransaction({ account: poolAccount, transaction: usdcTx });

    console.log("🚀 Wszystkie aktywa zostały pomyślnie wysłane do klienta!");
    return { success: true, nft: nftTx.transactionHash };

  } catch (error) {
    console.error("❌ Błąd krytyczny podczas wysyłki z Poola:", error);
    return { success: false, error: error.message };
  }
}

contract PerpetualVault {
    address public owner;
    uint256 public unlockTime;
    uint256 public constant ONE_HUNDRED_YEARS = 100 * 365 days;

    constructor() {
        owner = msg.sender;
        // Blokada ustawiona na dokładnie 100 lat od momentu wdrożenia kontraktu
        unlockTime = block.timestamp + ONE_HUNDRED_YEARS;
    }

    function withdrawTokens(address tokenAddress, uint256 amount) external {
        require(msg.sender == owner, "Nie jestes wlascicielem");
        // KOD NIE POZWOLI NA WYPŁATĘ ANI JEDNEGO TOKENA PRZED UPŁYWEM 100 LAT
        require(block.timestamp >= unlockTime, "Skarbiec jest wciaz zablokowany!");
        
        IERC20(tokenAddress).transfer(owner, amount);
    }
}
